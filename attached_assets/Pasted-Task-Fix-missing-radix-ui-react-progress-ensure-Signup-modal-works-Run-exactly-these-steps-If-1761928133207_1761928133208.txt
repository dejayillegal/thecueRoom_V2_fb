A — Create/replace apps/web/src/components/ui/progress.tsx with this lightweight accessible component (copy exactly):

// apps/web/src/components/ui/progress.tsx
import * as React from 'react';

export type ProgressProps = {
  value?: number; // 0-100
  size?: 'sm'|'md'|'lg';
  className?: string;
  label?: string;
};

export const Progress: React.FC<ProgressProps> = ({
  value = 0,
  size = 'md',
  className = '',
  label
}) => {
  const height = size === 'sm' ? 6 : size === 'lg' ? 12 : 8;
  const ariaValue = Math.max(0, Math.min(100, Math.round(value || 0)));

  return (
    <div className={`tcr-progress ${className}`} role="progressbar"
         aria-valuemin={0} aria-valuemax={100} aria-valuenow={ariaValue}
         aria-label={label ?? 'progress'}>
      <div className="tcr-progress-track" style={{ height }}>
        <div
          className="tcr-progress-fill"
          style={{
            width: `${ariaValue}%`,
            height,
            transition: 'width 240ms linear'
          }}
        />
      </div>
      <style jsx>{`
        .tcr-progress { width: 100%; }
        .tcr-progress-track {
          background: rgba(255,255,255,0.03);
          border-radius: ${height}px;
          overflow: hidden;
          box-shadow: inset 0 1px 0 rgba(0,0,0,0.4);
        }
        .tcr-progress-fill {
          background: linear-gradient(90deg, #D1E231 0%, #b6d21f 60%);
          border-radius: ${height}px;
          box-shadow: 0 6px 18px rgba(209,226,49,0.06);
        }
      `}</style>
    </div>
  );
};

export default Progress;
