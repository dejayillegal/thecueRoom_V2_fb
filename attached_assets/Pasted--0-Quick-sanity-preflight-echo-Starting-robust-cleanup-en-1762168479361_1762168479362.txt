/**
 * safe-fetch: wrapper around fetch with timeout, retries, and safe-json handling.
 */
import crossFetch from "cross-fetch";

export async function safeFetch(url, options = {}) {
  const {
    timeout = parseInt(process.env.NODE_FETCH_TIMEOUT_MS || "15000", 10),
    attempts = parseInt(process.env.POLL_RETRY_ATTEMPTS || "3", 10),
    headers = {},
    ...rest
  } = options;
  let lastErr = null;
  for (let i = 0; i < attempts; i++) {
    const controller = new AbortController();
    const id = setTimeout(() => controller.abort(), timeout);
    try {
      const res = await crossFetch(url, { headers, signal: controller.signal, ...rest });
      clearTimeout(id);
      const ct = res.headers.get("content-type") || "";
      const text = await res.text();
      if (ct.includes("application/json")) {
        try { return { ok: true, status: res.status, data: JSON.parse(text) }; } catch(e) {
          return { ok:false, status: res.status, error: "invalid-json", snippet: text.slice(0,500) };
        }
      } else if (ct.includes("text/") || ct.includes("html")) {
        // upstream returned HTML - not valid JSON
        return { ok:false, status: res.status, error: "html-response", snippet: text.slice(0,500) };
      } else {
        // binary or other - return raw buffer if needed
        return { ok:true, status: res.status, buffer: Buffer.from(text, "utf8") };
      }
    } catch (err) {
      lastErr = err;
      if (err.name === "AbortError") lastErr = new Error("timeout");
      // exponential backoff
      await new Promise(r => setTimeout(r, 200 * Math.pow(2, i)));
      continue;
    }
  }
  return { ok:false, status: 0, error: "fetch_failed", cause: String(lastErr) };
}

import { useEffect, useState } from "react";
/** useDebounce - returns debounced value */
export function useDebounce(value, delay = 300) {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);
  return debounced;
}
export default useDebounce;

/**
 * Minimal AI worker scaffold for Replit / local dev.
 * Supports job enqueue (file-based) and TEST_MODE behavior.
 */
const fs = require('fs');
const path = require('path');
const os = require('os');
const AI_TEMP_DIR = process.env.AI_TEMP_DIR || '/tmp/thecueroom-ai';
if (!fs.existsSync(AI_TEMP_DIR)) fs.mkdirSync(AI_TEMP_DIR, { recursive: true });

const concurrency = parseInt(process.env.AI_WORKER_CONCURRENCY || '1', 10);
let running = 0;
const queueFile = path.join(AI_TEMP_DIR, 'queue.json');
function readQueue(){ try { return JSON.parse(fs.readFileSync(queueFile,'utf8')||'[]'); } catch(e){ return []; } }
function writeQueue(q){ fs.writeFileSync(queueFile, JSON.stringify(q,null,2)); }

function enqueue(job){ const q = readQueue(); q.push(job); writeQueue(q); return job.jobId; }

async function workOne(job){
  const statusFile = path.join(AI_TEMP_DIR, job.jobId + '.status.json');
  fs.writeFileSync(statusFile, JSON.stringify({ jobId: job.jobId, status: 'running' }));
  try {
    if (job.type === 'fallbackSvgRasterize'){
      // create a tiny placeholder PNG (in TEST_MODE we just echo a PNG data URI)
      const out = path.join(AI_TEMP_DIR, job.jobId + '.png');
      // write a small black PNG placeholder
      fs.writeFileSync(out, Buffer.from('89504e470d0a1a0a0000000d49484452', 'hex'));
      fs.writeFileSync(statusFile, JSON.stringify({ jobId: job.jobId, status:'done', result: out }));
    } else if (job.type === 'verifyArtist'){
      // very rough heuristic: success if socialProfile is present
      const ok = !!(job.payload && job.payload.socialProfile);
      fs.writeFileSync(statusFile, JSON.stringify({ jobId: job.jobId, status: ok? 'verified' : 'pending_review', details: ok ? 'auto-verified-test' : 'requires-admin' }));
    } else {
      fs.writeFileSync(statusFile, JSON.stringify({ jobId: job.jobId, status:'unknown-type' }));
    }
  } catch(e){
    fs.writeFileSync(statusFile, JSON.stringify({ jobId: job.jobId, status:'error', error: String(e) }));
  }
}

async function processQueue(){
  const q = readQueue();
  if (!q.length) return;
  const job = q.shift();
  writeQueue(q);
  running++;
  await workOne(job);
  running--;
}

if (require.main === module){
  console.log('AI Worker started (TEST_MODE fallback). Concurrency:', concurrency);
  setInterval(() => {
    if (running < concurrency) {
      processQueue().catch(e => console.error(e));
    }
  }, 500);
}

module.exports = { enqueue };
NODE

import fs from 'fs';
import path from 'path';
const seedDir = path.join(process.cwd(), 'seeds', 'data');

export function loadSeed(name: 'users'|'threads'|'replies'|'events') {
  try {
    const p = path.join(seedDir, name + '.json');

    return [];
  } catch(e){ return []; }
}
export default loadSeed;
