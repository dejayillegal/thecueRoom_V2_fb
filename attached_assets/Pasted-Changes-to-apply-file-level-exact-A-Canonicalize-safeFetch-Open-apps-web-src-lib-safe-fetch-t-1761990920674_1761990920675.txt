// apps/web/src/lib/safe-fetch.ts
import fetch from 'cross-fetch';

export type SafeFetchOptions = {
  timeout?: number;
  attempts?: number;
  headers?: Record<string,string>;
  signal?: AbortSignal | null;
};

function sleep(ms:number){ return new Promise(r=>setTimeout(r,ms)); }

export async function safeFetch(url: string, options: SafeFetchOptions = {}) {
  const timeout = options.timeout ?? parseInt(process.env.NODE_FETCH_TIMEOUT_MS || '15000', 10);
  const attempts = options.attempts ?? parseInt(process.env.POLL_RETRY_ATTEMPTS || '3', 10);
  const headers = options.headers ?? {};
  const signal = options.signal ?? null;

  let lastError: any = null;
  for (let attempt = 1; attempt <= attempts; attempt++) {
    const controller = new AbortController();
    const to = setTimeout(() => controller.abort(), timeout);
    try {
      const res = await fetch(url, {
        method: 'GET',
        headers,
        signal: signal ?? controller.signal,
      });
      clearTimeout(to);
      if (!res.ok) {
        const text = await res.text().catch(()=>null);
        throw { status: res.status, statusText: res.statusText, body: text };
      }
      const contentType = res.headers.get('content-type') || '';
      // Let callers handle text/json parsing; return full response wrapper
      return { ok: true, response: res, contentType };
    } catch (err) {
      clearTimeout(to);
      lastError = err;
      const backoff = 300 * Math.pow(2, attempt - 1);
      if (attempt < attempts) await sleep(backoff);
    }
  }
  return { ok: false, error: lastError };
}

export default safeFetch;

B — Isolate Playwright into a server-only worker (no bundling)

Problem cause: a file under packages/feeds/* imports Playwright and is then imported (directly or indirectly) into Next's API route code that the bundler tries to compile for client. Fix by moving Playwright into a separate Node worker script that is never imported by the Next bundle.

Create a new worker file (server-only, outside Next bundle scope):

packages/feeds/headless-worker.js (Node, CommonJS) — NOT imported in client code.

// packages/feeds/headless-worker.js
const { fork } = require('child_process');
const path = require('path');
// This file is the entry point for a node worker that will spawn Playwright when needed.
// Use IPC: parent process will fork this worker with `child_process.fork` and send {id, url, selector}
// Worker will reply with { id, ok: true, html: '...', error?: ... }

const playwright = require('playwright'); // require at runtime

process.on('message', async (msg) => {
  try {
    if (!msg || !msg.url) return process.send({ id: msg.id, ok: false, error: 'no url' });
    const browser = await playwright.chromium.launch({ args: ['--no-sandbox', '--disable-setuid-sandbox'] });
    const page = await browser.newPage();
    await page.goto(msg.url, { waitUntil: 'networkidle', timeout: msg.timeout || 15000 });
    if (msg.waitForSelector) {
      await page.waitForSelector(msg.waitForSelector, { timeout: msg.waitForSelectorTimeout || 5000 }).catch(()=>{});
    }
    const html = await page.content();
    await browser.close();
    process.send({ id: msg.id, ok: true, html });
  } catch (err) {
    process.send({ id: msg.id, ok: false, error: err && err.message ? err.message : String(err) });
  }
});


Create a small server-side client utility to call the worker via child_process.fork() with a concurrency limiter. Place it in packages/feeds/headless.ts but do not import Playwright here; instead spawn the worker file above.

packages/feeds/headless.ts (Node, ESM or CJS — server-only usage):

import { fork } from 'child_process';
import path from 'path';
import pLimit from 'p-limit';

const PLAYWRIGHT_ENABLED = process.env.PLAYWRIGHT_ENABLED === 'true';
const WORKER_PATH = path.join(process.cwd(), 'packages/feeds/headless-worker.js');
const limit = pLimit(parseInt(process.env.PLAYWRIGHT_CONCURRENCY || '1', 10));

export async function renderPageWithHeadless(url: string, opts = {}) {
  if (!PLAYWRIGHT_ENABLED) {
    return { ok: false, error: 'PLAYWRIGHT_DISABLED' };
  }
  return limit(async () => {
    return new Promise((resolve) => {
      const child = fork(WORKER_PATH, { stdio: ['pipe', 'pipe', 'pipe', 'ipc'] });
      const id = `${Date.now()}-${Math.random().toString(36).slice(2,8)}`;
      const timer = setTimeout(() => {
        child.kill();
        resolve({ ok: false, error: 'timeout' });
      }, opts.timeout || 30000);
      child.on('message', (msg) => {
        if (msg && msg.id === id) {
          clearTimeout(timer);
          child.kill();
          resolve({ ok: msg.ok, html: msg.html, error: msg.error });
        }
      });
      child.send({ id, url, timeout: opts.timeout, waitForSelector: opts.waitForSelector, waitForSelectorTimeout: opts.waitForSelectorTimeout });
    });
  });
}
