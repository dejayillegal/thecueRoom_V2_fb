  id uuid PRIMARY KEY REFERENCES auth.users(id),
  display_name text NOT NULL,
  slug text UNIQUE,
  bio text,
  is_artist boolean DEFAULT false,
  artist_verified boolean DEFAULT false,
  spotlight boolean DEFAULT false,
  avatar_path text,
  socials jsonb DEFAULT '{}',
  created_at timestamptz DEFAULT now()
);

CREATE TABLE artist_spotlight (
  id bigserial PRIMARY KEY,

  weight integer DEFAULT 1,
  created_at timestamptz DEFAULT now()
);

-- News
CREATE TABLE news (
  id bigserial PRIMARY KEY,
  title text NOT NULL,
  content text NOT NULL,
  source text,
  url text,
  tags text[],
  published_at timestamptz,
  created_at timestamptz DEFAULT now()
);

-- Playlists / Tracks
CREATE TABLE playlists (
  id bigserial PRIMARY KEY,
  title text,
  description text,
  curator_id uuid REFERENCES profiles(id),

  platform_id text,
  embed_url text,
  created_at timestamptz DEFAULT now()

-- Forum
CREATE TABLE forums (
  id bigserial PRIMARY KEY,
  title text NOT NULL,
  description text,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE forum_threads (
  id bigserial PRIMARY KEY,
  forum_id bigint REFERENCES forums(id) ON DELETE CASCADE,
  author_id uuid REFERENCES profiles(id),
  title text NOT NULL,
  body text,
  pinned boolean DEFAULT false,
  locked boolean DEFAULT false,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz
);

CREATE TABLE forum_posts (
  id bigserial PRIMARY KEY,
  thread_id bigint REFERENCES forum_threads(id) ON DELETE CASCADE,
  author_id uuid REFERENCES profiles(id),
  body text,
  attachments jsonb DEFAULT '[]', -- store image keys
  created_at timestamptz DEFAULT now()
);

CREATE TABLE reactions (
  id bigserial PRIMARY KEY,
  target_type text NOT NULL, -- 'thread'|'post'|'profile'|'epk'|'meme'
  target_id bigint NOT NULL,
  user_id uuid REFERENCES profiles(id),
  reaction text, -- 'like'
  created_at timestamptz DEFAULT now(),
  UNIQUE (target_type, target_id, user_id)
);

CREATE TABLE views (
  id bigserial PRIMARY KEY,
  target_type text NOT NULL,
  target_id bigint NOT NULL,
  user_id uuid,
  created_at timestamptz DEFAULT now()
);

-- Events + Tickets + RSVPs
CREATE TABLE events (
  id bigserial PRIMARY KEY,
  title text NOT NULL,
  organizer_id uuid REFERENCES profiles(id),
  description text,
  start_time timestamptz,
  end_time timestamptz,
  venue jsonb,
  flyer_path text,
  ticketing jsonb DEFAULT '{}', -- contains ticket types, external_url etc
  auto_approved boolean DEFAULT false,
  approved boolean DEFAULT false,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE tickets (
  id bigserial PRIMARY KEY,
  event_id bigint REFERENCES events(id) ON DELETE CASCADE,

  stock integer DEFAULT 0,
  created_at timestamptz DEFAULT now()

CREATE TABLE rsvps (
  id bigserial PRIMARY KEY,
  event_id bigint REFERENCES events(id),
  user_id uuid REFERENCES profiles(id),

  id bigserial PRIMARY KEY,

// FILE: apps/web/app/api/signup/route.ts
import { NextResponse } from 'next/server';
import { z } from 'zod';
import { createClient } from '@supabase/supabase-js';
import { ProfileCreate } from '../../../packages/shared/schemas';

const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export async function POST(req: Request) {
  const body = await req.json();
  const parse = ProfileCreate.safeParse(body);
  if (!parse.success) return NextResponse.json({ error: parse.error.format() }, { status: 400 });
  const { display_name, bio, is_artist, socials } = parse.data;

  // Create profile row — user must exist in Supabase Auth (post-auth hook)
  // For simplicity assume user is authenticated and we have user id from header
  const userId = req.headers.get('x-user-id');
  if (!userId) return NextResponse.json({ error: 'Missing user id' }, { status: 401 });

  const upsert = await supabase.from('profiles').upsert({
    id: userId,
    display_name,
    bio,
    is_artist: !!is_artist,
    socials
  }).select().single();

  if (upsert.error) return NextResponse.json({ error: upsert.error.message }, { status: 500 });
  return NextResponse.json({ profile: upsert.data });
}

'use client'
import React, { useState } from 'react';
import { createClient } from '@supabase/supabase-js';
import { useRouter } from 'next/navigation';

const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!);

export default function SignupModal({ userId }: { userId: string }) {
  const [displayName, setDisplayName] = useState('');
  const [bio, setBio] = useState('');
  const [isArtist, setIsArtist] = useState(false);
  const router = useRouter();

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    const res = await fetch('/api/signup', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-user-id': userId },
      body: JSON.stringify({ display_name: displayName, bio, is_artist: isArtist })
    });
    const data = await res.json();
    if (res.ok) router.refresh();
    else alert('Signup failed: ' + JSON.stringify(data));
  }

  return (
    <div className="p-6 bg-surface rounded-lg shadow-lg w-full max-w-md">
      <h3 className="text-xl font-semibold">Complete your profile</h3>
      <form onSubmit={submit} className="space-y-4 mt-4">
        <input value={displayName} onChange={e => setDisplayName(e.target.value)} placeholder="Display name" className="w-full p-2 bg-black border rounded" />
        <textarea value={bio} onChange={e => setBio(e.target.value)} placeholder="Bio" className="w-full p-2 bg-black border rounded"></textarea>
        <label className="flex items-center gap-2">
          <input type="checkbox" checked={isArtist} onChange={e => setIsArtist(e.target.checked)} />
          <span>Sign up as Artist / DJ</span>
        </label>
        <button className="btn btn-primary">Save profile</button>
      </form>
    </div>
  );
}

// FILE: packages/server/aiWorker.ts
import { createClient } from '@supabase/supabase-js';
import fetch from 'node-fetch'; // or global fetch
const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export async function processNextJob() {
  const { data: jobs } = await supabase.from('ai_jobs').select('*').eq('status', 'queued').limit(1).order('created_at', { ascending: true });
  if (!jobs || jobs.length === 0) return;
  const job = jobs[0];
  await supabase.from('ai_jobs').update({ status: 'running' }).eq('id', job.id);

  try {
    // Example: call OpenAI Image API (DALL·E or images) or a local model
    const prompt = job.input?.prompt || 'abstract techno cover';
    const aiRes = await fetch('https://api.openai.com/v1/images/generations', {
      method: 'POST',
      headers: { 'Authorization': `Bearer ${process.env.OPENAI_API_KEY}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt, size: '1024x1024', n: 1 })
    });
    const payload = await aiRes.json();
    // save image to supabase storage
    // ... omitted: decode base64, upload file, set result_path
    await supabase.from('ai_jobs').update({ status: 'done', result_path: '/storage/path.png' }).eq('id', job.id);
  } catch (err) {
    await supabase.from('ai_jobs').update({ status: 'failed' }).eq('id', job.id);
  }
}
export default processNextJob;

// FILE: apps/web/app/api/events/submit/route.ts
import { NextResponse } from 'next/server';
import { EventSubmit } from '../../../../packages/shared/schemas';
import { createClient } from '@supabase/supabase-js';
const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export async function POST(req: Request) {
  const body = await req.json();
  const parse = EventSubmit.safeParse(body);
  if (!parse.success) return NextResponse.json({ error: parse.error }, { status: 400 });

  const event = parse.data;
  // AI validation (call OpenAI)
  const aiCheck = await fetch('https://api.openai.com/v1/chat/completions', {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${process.env.OPENAI_API_KEY}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({
      model: 'gpt-4o-mini', messages: [{ role: 'system', content: 'You are a moderator. Determine if this event submission is valid and not spam or self-promo. Return JSON: {approve: true|false, reason:""}' }, { role: 'user', content: JSON.stringify(event) }]
    })
  });
  const aiJson = await aiCheck.json();
  const approve = aiJson?.choices?.[0]?.message?.content?.includes('"approve": true') || false;
  const { data: created } = await supabase.from('events').insert([{ ...event, auto_approved: approve, approved: approve }]).select().single();
  // If not approved send admin notification
  if (!approve) {
    await supabase.from('notifications').insert([{ user_id: /* admin id or admin role */ null, payload: { type: 'event_review', event_id: created.id } }]);
  }
  return NextResponse.json({ ok: true, event: created });
}

import { NextResponse } from 'next/server';
import QRCode from 'qrcode';
import { createClient } from '@supabase/supabase-js';
const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export async function POST(req: Request) {
  const body = await req.json();
  const { event_id, user_id, ticket_id } = body;
  const token = Math.random().toString(36).slice(2) + Date.now().toString(36);
  const dataUri = await QRCode.toDataURL(token);
  const { data } = await supabase.from('rsvps').insert([{ event_id, user_id, ticket_id, qr_code: dataUri }]).select().single();
  // Optionally upload QR image to Supabase Storage for long-term
  return NextResponse.json({ ok: true, rsvp: data, qr_token: token, qr_data_uri: dataUri });
}

// FILE: apps/web/app/api/ticket/scan/route.ts
import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';
const supabase = createClient(process.env.NEXT_PUBLIC_SUPABASE_URL!, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export async function POST(req: Request) {
  const { token } = await req.json();
  // Find rsvp with matching token in QR or search by qr_code string
  const { data } = await supabase.from('rsvps').select('*').like('qr_code', `%${token}%`).limit(1).single();
  if (!data) return NextResponse.json({ ok: false, error: 'Not found' }, { status: 404 });
  // Mark attendance (add field 'checked_in' later)
  await supabase.from('rsvps').update({ checked_in: true, checked_in_at: new Date().toISOString() }).eq('id', data.id);
  return NextResponse.json({ ok: true, rsvp: data });
}
